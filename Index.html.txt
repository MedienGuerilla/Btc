  <script>
    const $ = (id) => document.getElementById(id);
    const money = (value) => new Intl.NumberFormat("de-DE", {
      style: "currency", currency: "USD", maximumFractionDigits: 0
    }).format(value);
    const compact = (value) => new Intl.NumberFormat("de-DE", {
      notation: "compact", maximumFractionDigits: 2
    }).format(value);

    async function loadData() {
      try {
        const [marketRes, fearRes, chainRes, chartRes] = await Promise.all([
          fetch("https://api.coingecko.com/api/v3/coins/bitcoin?localization=false&tickers=false&market_data=true&community_data=false&developer_data=false"),
          fetch("https://api.alternative.me/fng/?limit=1"),
          fetch("https://blockchain.info/q/hashrate"),
          fetch("https://api.coingecko.com/api/v3/coins/bitcoin/market_chart?vs_currency=usd&days=30")
        ]);

        if (!marketRes.ok || !fearRes.ok || !chainRes.ok || !chartRes.ok) {
          throw new Error("Eine Datenquelle ist momentan nicht erreichbar.");
        }

        const market = await marketRes.json();
        const fear = await fearRes.json();
        const hashrate = await chainRes.json();
        const chart = await chartRes.json();
        const data = market.market_data;
        const usd = data.current_price.usd;

        $("price").textContent = money(usd);
        const change = data.price_change_percentage_24h;
        $("change").textContent = `${change >= 0 ? "+" : ""}${change.toFixed(2)}% in 24 Std.`;
        $("change").className = `change ${change >= 0 ? "positive" : "negative"}`;

        $("marketCap").textContent = "$" + compact(data.market_cap.usd);
        $("volume").textContent = "$" + compact(data.total_volume.usd);

        const fearValue = Number(fear.data[0].value);
        $("fear").textContent = `${fearValue} / 100`;
        $("fearLabel").textContent = fear.data[0].value_classification;
        $("needle").style.left = `${fearValue}%`;

        const hashesPerSecond = Number(hashrate) * 1e12;
        $("hashrate").textContent = Number.isFinite(hashesPerSecond)
          ? `${compact(hashesPerSecond / 1e18)} EH/s` : "—";

        const prices = chart.prices.map(point => point[1]);
        $("support").textContent = money(Math.min(...prices));
        $("resistance").textContent = money(Math.max(...prices));

        $("updated").textContent = `Aktualisiert: ${new Date().toLocaleTimeString("de-DE", {
          hour: "2-digit", minute: "2-digit"
        })}`;
      } catch (error) {
        $("updated").textContent = "Daten momentan nicht verfügbar";
        $("change").textContent = "Bitte später erneut versuchen.";
        console.error(error);
      }
    }

    loadData();
    setInterval(loadData, 60_000);
  </script>
